#!/usr/bin/env bash
# D1 probe: the shipped tutorial generic_TRISO_1D (mechanics, TRISO gap model, SCIANTIX, PyC swelling and creep all on), on a copy.
# The tutorial aborts at its last step because its heat table ends exactly at the end time (probe T0); here one more point is
# appended beyond the end time, so the run goes through. usage: bash run-tutorial.sh.txt [timeout_seconds]
set +eu
source /opt/openfoam9/etc/bashrc
set +e
TO="${1:-900}"
SRC=/home/feed/OpenFOAM/feed-9/applications/solvers/offbeat/Cases/testCases/generic_TRISO_1D
RUN="$HOME/spike/d1/tutorial"
rm -rf "$RUN"; mkdir -p "$RUN"; cp -r "$SRC"/. "$RUN"/; cd "$RUN" || exit 3
rm -rf Output
grep -n "timePoints\|vhgr" constant/solverDict
sed -i 's/timePoints  (0.0 3600 49996400 5e7);/timePoints  (0.0 3600 49996400 5e7 6e7);/; s/vhgr        (0.0 3e9  3e9      0.0);/vhgr        (0.0 3e9  3e9      0.0 0.0);/' constant/solverDict
grep -n "timePoints\|vhgr" constant/solverDict
s=$(date +%s.%N)
timeout 120 blockMesh > log.blockMesh 2>&1; echo "blockMesh exit $?"
[ -f system/changeDictionaryDict ] && { timeout 120 changeDictionary > log.changeDictionary 2>&1; echo "changeDictionary exit $?"; }
timeout "$TO" offbeat > log.offbeat 2>&1; rc=$?; echo "offbeat exit $rc"
e=$(date +%s.%N); echo "wall $(echo "$e - $s" | bc) s"
ls -d [0-9]* 2>/dev/null | sort -g | tail -3 | tr '\n' ' '; echo
tail -4 log.offbeat | cut -c1-200
grep -i "Gas pressure" log.offbeat | tail -1
timeout 120 postProcess -func writeCellCentres -latestTime > log.cc 2>&1; echo "writeCellCentres exit $?"
