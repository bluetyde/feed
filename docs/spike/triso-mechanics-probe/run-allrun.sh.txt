#!/usr/bin/env bash
# D1 probe: run a shipped case with its OWN Allrun script, on a copy under ~/spike/d1/allrun/<case>, and print its results file.
# usage: bash run-allrun.sh <case path below Cases/, or just a trisoBenchmarks case name> [timeout_seconds]
#   bash run-allrun.sh case_9 600        bash run-allrun.sh Verification/trisoModulesTest/hollowsphere_1D 300
set +eu
source /opt/openfoam9/etc/bashrc
set +e
P="$1"; TO="${2:-900}"
case "$P" in */*) ;; *) P="Verification/trisoBenchmarks/$P" ;; esac
C=$(basename "$P")
SRC=/home/feed/OpenFOAM/feed-9/applications/solvers/offbeat/Cases/$P
RUN="$HOME/spike/d1/allrun/$C"
rm -rf "$RUN"; mkdir -p "$RUN"; cp -r "$SRC"/. "$RUN"/; cd "$RUN" || exit 3
rm -f results
s=$(date +%s.%N)
timeout "$TO" bash ./Allrun > log.Allrun 2>&1; rc=$?
e=$(date +%s.%N)
echo "$C: Allrun exit $rc, $(echo "$e - $s" | bc) s"
ls -d [0-9]* 2>/dev/null | sort -g | tr '\n' ' '; echo
cat results 2>/dev/null
source /opt/openfoam9/etc/bashrc
timeout 120 postProcess -func writeCellCentres -latestTime > log.cc 2>&1
